// actions/pending.ts
export function createPending(a: ProposedAction, discordUserId: string, now: Date): Promise<PendingAction>;
export function takePending(id: string, discordUserId: string, now: Date): Promise<PendingAction | "expired" | "not_yours">;

BEHAVIOR: pending.ts
1. createPending: id = 12 characters from `[A-Za-z0-9]` via `crypto.getRandomValues`;
   `expiresAt = now + 600000`; store at `keys.pending(id)` with TTL 600; return the record.
2. takePending: missing, or `expiresAt <= now` -> "expired". Different `discordUserId` ->
   "not_yours", and the record is left exactly as it was. Otherwise delete it and return it
   (single use).
3. saveThread: same id scheme; `createdAt = now`, `expiresAt = now + 600000`; TTL 600 at
   `keys.thread(id)`.
4. loadThread: missing or expired -> "expired"; other user -> "not_yours" (untouched);
   otherwise set `expiresAt = now + 600000`, write it back with TTL 600 (sliding), return it.

BEHAVIOR: execute.ts
5. WRITE_ARGS: zod `.strict()` objects (a `tenant` key, or any unknown key, is rejected; the
   tenant is injected later by callTool). Ids are positive integers. These match the live
   Wizzrobe MCP input schemas:
   - acknowledge_match, offer_match_for_stream, withdraw_stream_offer, watch_match,
     unwatch_match: `{match_id}`
   - signup_as_crew, withdraw_crew_signup: `{match_id, role: "commentator" | "tracker"}`
   - sign_up_for_tournament, withdraw_from_tournament: `{tournament_id}`
   - submit_match_request: `{tournament_id, scheduled_date: /^\d{4}-\d{2}-\d{2}$/,
     scheduled_time: /^([01]\d|2[0-3]):[0-5]\d$/, player_ids: int[] (at least 2),
     comment?: string up to 500}` (date and time are on the community clock)
   - rest_create_reschedule_request: `{match_id, kind: "reschedule" | "cancel", reason:
     string 1..500, proposed_at?: ISO 8601 UTC string}`
   - rest_book_bracket_game: `{bracket_match_id, scheduled_date, scheduled_time, comment?}`
     with the same date and time patterns.
6. executeAction(p): validate `p.args` with `WRITE_ARGS[p.tool]`; failure ->
   `UserInputError("That request no longer looks right. Ask again.")` and log warn with the
   tool name and zod issue paths only. `token = await userAccessToken(p.discordUserId)` (never
   the bot's). Then:
   - `rest_create_reschedule_request` -> `createRescheduleRequest(token, match_id, {kind,
     reason, proposedAt: proposed_at})`;
   - `rest_book_bracket_game` -> `bookBracketGame(token, bracket_match_id, {scheduledDate,
     scheduledTime, comment})`;
   - every other tool -> `callTool(token, p.tool, args)`.
   Errors propagate unchanged (L16). Log info "action.executed" `{tool, ok}`.
7. Result line, plain words, no dashes:
   - acknowledge_match: "Acknowledged match <id>."
   - offer_match_for_stream: "Offered match <id> for stream. Stream managers decide what goes
     on air."; withdraw_stream_offer: "Withdrew your stream offer for match <id>."
   - watch_match: "You'll get Wizzrobe DMs when match <id> changes."; unwatch_match: "Stopped
     watching match <id>."
   - signup_as_crew: "Signed you up as <role> for match <id>. Staff may need to approve it.";
     withdraw_crew_signup: "Withdrew your <role> signup for match <id>."
   - sign_up_for_tournament: "Entered you in tournament <id>."; withdraw_from_tournament:
     "Withdrew you from tournament <id>."
   - submit_match_request: "Sent your match request for <date> <time>. Staff will approve or
     decline it."
   - rest_create_reschedule_request: "Sent your <kind> request for match <id>. Staff will
     decide."; rest_book_bracket_game: "Booked the next game for <date> <time>."
