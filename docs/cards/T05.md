```ts
// store/kv.ts
export interface Kv {
  get<T>(key: string): Promise<T | null>;
  set<T>(key: string, value: T, ttlSec?: number): Promise<void>;
  del(key: string): Promise<void>;
  incrByFloat(key: string, by: number, ttlSec?: number): Promise<number>;
  expire(key: string, ttlSec: number): Promise<void>;
}
export function kv(): Kv;                        // Upstash, lazy singleton
export function setKvForTests(k: Kv | null): void;

// store/keys.ts
export const keys: {
  link(discordUserId: string): string;           // LinkRecord JSON (its refresh token is sealed)
  oauthState(state: string): string;             // PKCE verifier + discord user, 10 min
  botToken(): string;                            // cached bot access token
  userToken(discordUserId: string): string;      // cached user access token
  pending(id: string): string;                   // PendingAction
  thread(id: string): string;                    // disambiguation / follow-up context
  spend(day: string): string;                    // daily USD
  cache(scope: string, hash: string): string;    // short-lived Wizzrobe read cache
};

// store/crypto.ts
export function seal(plaintext: string): Promise<string>;     // AES-256-GCM, TOKEN_ENCRYPTION_KEY
export function open(sealed: string): Promise<string>;
```

BEHAVIOR
1. keys: every key starts `wb:v1:`. Exact forms: `wb:v1:link:<id>`, `wb:v1:oauth:<state>`,
   `wb:v1:token:bot`, `wb:v1:token:user:<id>`, `wb:v1:pending:<id>`, `wb:v1:thread:<id>`,
   `wb:v1:spend:<YYYY-MM-DD>`, `wb:v1:cache:<scope>:<hash>`. Validate inputs and throw a plain
   Error on a bad one: discord ids are digits; state, pending, thread ids, scope and hash match
   `^[A-Za-z0-9_.-]{1,128}$`; day matches `^\d{4}-\d{2}-\d{2}$`.
2. kv: `import { Redis } from "@upstash/redis"`;
   `new Redis({ url: config().kvRestApiUrl, token: config().kvRestApiToken })`, created on first
   use.
   - get -> `redis.get<T>(key)` (the client JSON-decodes);
   - set -> `redis.set(key, value, { ex: ttlSec })` when ttlSec is given, else `redis.set(key,
     value)`;
   - del -> `redis.del(key)`;
   - incrByFloat -> `redis.incrbyfloat(key, by)`, then `redis.expire(key, ttlSec)` when given;
     returns the new value as a number;
   - expire -> `redis.expire(key, ttlSec)`.
   Errors from Upstash propagate unchanged (L16). `setKvForTests(k)` swaps the singleton; null
   resets it so the next `kv()` builds a real client.
3. crypto: Web Crypto (`globalThis.crypto.subtle`). Key = base64-decode of
   `config().tokenEncryptionKey` (32 bytes), imported once as AES-GCM. seal: random 12-byte iv;
   output `base64(iv) + "." + base64(ciphertext with tag)`. open: exactly two non-empty parts,
   iv 12 bytes, decrypt; any failure throws `Error("sealed value is invalid")` (catch and
   rethrow, never an empty catch). Never log plaintext.

  ciphertext part throws; a wrong key throws (set a different 32-byte base64

  number, an Upstash rejection propagating, and setKvForTests swapping and resetting.
